# Выгрузка по дням и nm_id

### 1) Создание таблицы с данными

Создание таблиц, куда будут записаны необходимые данные на каждом сервере с 1 по 14

```sql
CREATE TABLE IF NOT EXISTS support_schema.wb_02_manual_analytics_by_subject_name_all_clients (
    client_id BIGINT NOT NULL
    , subject_name VARCHAR NOT NULL
    , nm_ids_count BIGINT
    , clicks_auto NUMERIC
    , clicks_keywords NUMERIC
    , views_auto NUMERIC
    , views_keywords NUMERIC
    , clicks_search NUMERIC
    , clicks_keywords_search NUMERIC
    , views_search NUMERIC
    , views_keywords_search NUMERIC
    , add_to_cart_count NUMERIC
    , open_card_count NUMERIC
    , orders_count NUMERIC
)
;
ALTER TABLE support_schema.wb_02_manual_analytics_by_subject_name_all_clients ADD CONSTRAINT wb_02_manual_analytics_by_subject_name_all_clients_pkey PRIMARY KEY (client_id, subject_name)
;
```

### 2) Запуск сбора аналитики

Запускать в DBeaver для серверов с 1 по 14. Можно запускать параллельно.
Не обрабатываются клиенты, у которых больше 5000 артикулов.

```sql
SET statement_timeout = '45min'
;

-- ФУНКЦИЯ ЗАПУСКА
DO $$
DECLARE
    current_schema_name text;
    final_sql text;
	has_all_tables boolean;
    nm_id_count integer;
    sql text;
BEGIN
	RAISE NOTICE '                            ';
	RAISE NOTICE 'new func start';
	RAISE NOTICE '                            ';

    FOR current_schema_name IN 
        SELECT schema_name
        FROM information_schema.schemata
        WHERE schema_name ~ '^schema_\d+$'
        ORDER BY CAST(substring(schema_name FROM '\d+$') AS INTEGER)
        LIMIT 1000
        OFFSET 0
    LOOP
	BEGIN

        sql := format('SELECT COUNT(DISTINCT nm_id) FROM %I.wb_cards', current_schema_name);
        EXECUTE sql INTO nm_id_count;

        IF nm_id_count > 5000 THEN
            RAISE NOTICE 'Schema % skipped: more than 5000 distinct nm_id in wb_cards', current_schema_name;
            CONTINUE;
        END IF;

       SELECT bool_and(exists_flag) INTO has_all_tables
        FROM (
            VALUES 
                ('wb_cards'),
                ('ds_wb10x_checklist_by_days_and_nm_ids_v1')
        ) t(tbl)
        LEFT JOIN LATERAL (
            SELECT EXISTS (
                SELECT 1 
                FROM information_schema.tables 
                WHERE table_schema = current_schema_name 
                  AND table_name = t.tbl
            ) AS exists_flag
        ) chk ON true;

        IF NOT has_all_tables THEN
            RAISE NOTICE 'Skipping schema % because not all required tables exist', current_schema_name;
            CONTINUE;
        END IF;

        final_sql := format($f$


            WITH
                nm_ids_subject_names AS (
                    SELECT
                        nm_id
                        , COALESCE(subject_name, '-') AS subject_name
                    FROM %I.wb_cards w1
                    WHERE updated_at = (
                        SELECT MAX(updated_at)
                        FROM %I.wb_cards w2
                        WHERE w2.nm_id = w1.nm_id
                    )
                )
                , dates_filter AS (
                    SELECT 1
                        , CURRENT_DATE - 30 AS date_from
                        , CURRENT_DATE - 1 AS date_to
                )
                , nm_ids AS (
                    SELECT DISTINCT nm_id FROM nm_ids_subject_names
                )
                , b AS (
                    SELECT
                        generate_series((SELECT date_from FROM dates_filter), (SELECT date_to FROM dates_filter), '1 day'::interval)::DATE AS date
                        , nm_id
                    FROM nm_ids
                )
                , b_subject_names AS (
                    SELECT
                        generate_series((SELECT date_from FROM dates_filter), (SELECT date_to FROM dates_filter), '1 day'::interval)::DATE AS date
                        , subject_name
                    FROM (select DISTINCT subject_name FROM nm_ids_subject_names)
                )
                , result AS (
                    SELECT
                        SUBSTRING(%L, 8)::BIGINT AS client_id
                        , nmsn.subject_name
                        , COUNT(DISTINCT d.nm_id) AS nm_ids_count

                        , SUM(d.clicks_auto) AS clicks_auto
                        , SUM(d.clicks_keywords) AS clicks_keywords
                        , SUM(d.views_auto) AS views_auto
                        , SUM(d.views_keywords) AS views_keywords

                        , SUM(d.clicks_search) AS clicks_search
                        , SUM(d.clicks_keywords_search) AS clicks_keywords_search
                        , SUM(d.views_search) AS views_search
                        , SUM(d.views_keywords_search) AS views_keywords_search

                        , SUM(d.add_to_cart_count) AS add_to_cart_count
                        , SUM(d.open_card_count) AS open_card_count
                        , SUM(d.orders_count) AS orders_count
                    FROM
                        b
                        LEFT JOIN %I.ds_wb10x_checklist_by_days_and_nm_ids_v1 d ON d.nm_id = b.nm_id AND d.date = b.date
                        LEFT JOIN nm_ids_subject_names nmsn ON b.nm_id = nmsn.nm_id
                    GROUP BY nmsn.subject_name
                )
                -- SELECT COUNT(*) FROM result
                -- ;
                INSERT INTO support_schema.wb_02_manual_analytics_by_subject_name_all_clients (
                    client_id
                    , subject_name
                    , nm_ids_count
                    , clicks_auto
                    , clicks_keywords
                    , views_auto
                    , views_keywords
                    , clicks_search
                    , clicks_keywords_search
                    , views_search
                    , views_keywords_search
                    , add_to_cart_count
                    , open_card_count
                    , orders_count
                )
                SELECT * FROM result
                ON CONFLICT DO NOTHING
                
        $f$, current_schema_name, current_schema_name, current_schema_name, current_schema_name, current_schema_name);
		
        RAISE NOTICE 'Executing for %', current_schema_name;
--		RAISE NOTICE 'SQL = %', final_sql;

        EXECUTE final_sql;
		
	    EXCEPTION
	        WHEN OTHERS THEN
	            RAISE NOTICE 'Skipping schema % due to error: %', current_schema_name, SQLERRM;
	END;
    END LOOP;
END;
$$ LANGUAGE plpgsql;
```

### 4) Создание таблицы с аналитикой по CTR на локальном сервере
```sql
-- Таблица предназначена для сохранения финальной аналитики по CTR
CREATE TABLE IF NOT EXISTS support_schema.wb_02_analytics_by_subject_name (
    subject_name VARCHAR NOT NULL

    , ctr_rs_cat NUMERIC
    , ctr_rs_cat_search NUMERIC
    , ctr_keywords NUMERIC
    , ctr_keywords_search NUMERIC
    , add_to_cart_conversion NUMERIC
    , click_to_order_conversion NUMERIC
)
;
ALTER TABLE support_schema.wb_02_analytics_by_subject_name ADD CONSTRAINT wb_02_analytics_by_subject_name_pkey PRIMARY KEY (subject_name)
;

WITH
    -- Ограничение на количество кликов. Если кликов меньше заданного значения - принимать их за 0
    clicks_limit AS (
        SELECT 50::NUMERIC AS clicks_limit
    )
    , sum_method AS (
        select
            subject_name
            , COALESCE(SUM(nm_ids_count), 0) AS nm_ids_count
            
            , COALESCE(SUM(clicks_auto), 0) AS clicks_auto
            , COALESCE(SUM(clicks_keywords), 0) AS clicks_keywords
            , COALESCE(SUM(views_auto), 0) AS views_auto
            , COALESCE(SUM(views_keywords), 0) AS views_keywords

            , COALESCE(SUM(clicks_search), 0) AS clicks_search
            , COALESCE(SUM(clicks_keywords_search), 0) AS clicks_keywords_search
            , COALESCE(SUM(views_search), 0) AS views_search
            , COALESCE(SUM(views_keywords_search), 0) AS views_keywords_search

            , COALESCE(SUM(add_to_cart_count), 0) AS add_to_cart_count
            , COALESCE(SUM(open_card_count), 0) AS open_card_count
            , COALESCE(SUM(orders_count), 0) AS orders_count
        FROM
            wb_02_manual_analytics_by_subject_name_total_all_clients
        GROUP BY subject_name
    )
    , sums_with_limits AS (
        SELECT
            subject_name
            , nm_ids_count

            , CASE WHEN clicks_auto > cl.clicks_limit THEN clicks_auto ELSE 0 END AS clicks_auto
            , CASE WHEN clicks_keywords > cl.clicks_limit THEN clicks_keywords ELSE 0 END AS clicks_keywords
            , views_auto
            , views_keywords

            , CASE WHEN clicks_search > cl.clicks_limit THEN clicks_search ELSE 0 END AS clicks_search
            , CASE WHEN clicks_keywords_search > cl.clicks_limit THEN clicks_keywords_search ELSE 0 END AS clicks_keywords_search
            , views_search
            , views_keywords_search

            , add_to_cart_count
            , open_card_count
            , orders_count
        FROM
            sum_method
            LEFT JOIN clicks_limit cl ON TRUE
    )
    , avg_sums AS (
        SELECT
            subject_name
            , nm_ids_count
            , clicks_auto / 30 AS clicks_auto
            , clicks_keywords / 30 AS clicks_keywords
            , views_auto / 30 AS views_auto
            , views_keywords / 30 AS views_keywords
            , clicks_search / 30 AS clicks_search
            , clicks_keywords_search / 30 AS clicks_keywords_search
            , views_search / 30 AS views_search
            , views_keywords_search / 30 AS views_keywords_search
            , add_to_cart_count / 30 AS add_to_cart_count
            , open_card_count / 30 AS open_card_count
            , orders_count / 30 AS orders_count
        FROM sums_with_limits
    )
    , total_values AS (
        SELECT
            d.subject_name

            , CASE WHEN d.nm_ids_count >= 2 THEN
                ROUND(COALESCE((d.clicks_auto - d.clicks_keywords) / NULLIF((d.views_auto  - d.views_keywords),  0), 0), 5)
            ELSE 0 END AS ctr_rs_cat

            , CASE WHEN d.nm_ids_count >= 2 THEN
                ROUND(COALESCE((d.clicks_search - d.clicks_keywords_search) / NULLIF((d.views_search  - d.views_keywords_search),  0), 0), 5)
            ELSE 0 END AS ctr_rs_cat_search

            , CASE WHEN d.nm_ids_count >= 2 THEN
                ROUND(COALESCE(d.clicks_keywords / NULLIF(d.views_keywords,  0), 0), 5)
            ELSE 0 END AS ctr_keywords

            , CASE WHEN d.nm_ids_count >= 2 THEN
                ROUND(COALESCE(d.clicks_keywords_search / NULLIF(d.views_keywords_search,  0), 0), 5)
            ELSE 0 END AS ctr_keywords_search

            , ROUND(COALESCE(d.add_to_cart_count/NULLIF(d.open_card_count, 0), 0), 5) AS add_to_cart_conversion
            , ROUND(COALESCE(d.orders_count/ NULLIF(d.add_to_cart_count, 0),0), 5) AS click_to_order_conversion
        FROM
            avg_sums d
    )
    , result AS (
        SELECT * FROM total_values
        WHERE subject_name IS NOT NULL AND subject_name <> '-' AND (
            FALSE
            OR ctr_rs_cat > 0
            OR ctr_rs_cat_search > 0
            OR ctr_keywords > 0
            OR ctr_keywords_search > 0
            OR add_to_cart_conversion > 0
            OR click_to_order_conversion > 0
        )
        order by subject_name
    )
    INSERT INTO support_schema.wb_02_analytics_by_subject_name (
        subject_name
        , ctr_rs_cat
        , ctr_rs_cat_search
        , ctr_keywords
        , ctr_keywords_search
        , add_to_cart_conversion
        , click_to_order_conversion
    )
    select * from result
    ;
```
